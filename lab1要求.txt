进阶要求（1分）：观察 AI 编译器 MLIR 中端处理过程

程序样例：
阶乘
main()
{
	int i, n, f;
 
	cin >> n;
	i = 2;
	f = 1;
	while (i <= n)
	{
		f = f * i;
		i = i + 1;
	}
	cout << f << endl;
}


斐波那契数列
main()
{
	int a, b, i, t, n;
 
	a = 0;
	b = 1;
	i = 1;
	cin >> n;
	cout << a << endl;
	cout << b << endl;
	while (i < n)
	{
		t = b;
		b = a + b;
		cout << b << endl;
		a = t;
		i = i + 1;
	}
}
